    ```typescript
    /**
     * Poll analysis status until completed/failed, with timeout.
     * This catches the case where the Agent never calls slide_complete_analysis.
     * nanobot-equivalent: runner.py _run_core's max_iterations fallback
     */
    async waitForCompletion(analysisId: number, timeoutMs: number = 120_000): Promise<AiAnalysisRecord | null> {
      const pollInterval = 2000; // 2s
      const deadline = Date.now() + timeoutMs;

      while (Date.now() < deadline) {
        const record = await this.getAnalysisById(analysisId);
        if (!record) return null;
        if (record.status === 'completed') return record;
        if (record.status === 'failed') return record;
        await new Promise(resolve => setTimeout(resolve, pollInterval));
      }

      // Timeout: auto-fail
      await this.failAnalysis(analysisId, '分析超时：Agent 未在规定时间内完成');
      // Read the final record after fail
      return this.getAnalysisById(analysisId);
    }
    ```

    Find the existing polling code (around line 275). Replace with:
    ```typescript
    private _stopDiagnosisPolling() {
      if (this.diagnosisPollTimer) {
        clearInterval(this.diagnosisPollTimer);
        this.diagnosisPollTimer = null;
      }
    }

    private _startPolling(analysisId: number) {
      this._stopDiagnosisPolling();
      this.diagnosisPollingStart = Date.now();

      this.diagnosisPollTimer = setInterval(async () => {
        try {
          const res = await authFetch(`/api/ai/analysis/status/${analysisId}`);
          const data = await res.json();
          if (!data.ok || !data.record) return;

          const record = data.record;

          if (record.status === "completed") {
            this.diagnosisStatus = "completed";
            this.diagnosisResult = record.result;
            this._stopDiagnosisPolling();
            // Refresh diagnosis history
            this._loadDiagnosisHistory();
          } else if (record.status === "failed") {
            this.diagnosisStatus = "failed";
            this.diagnosisError = record.error_message || "分析失败";
            this._stopDiagnosisPolling();
          }
          this.requestUpdate();
        } catch (err) {
          // Network error, keep polling
        }

        // Safety: stop after 3 minutes
        if (Date.now() - this.diagnosisPollingStart > 180_000) {
          this.diagnosisStatus = "failed";
          this.diagnosisError = "诊断超时";
          this._stopDiagnosisPolling();
          this.requestUpdate();
        }
      }, 2000); // Poll every 2 seconds
    }
    ```
